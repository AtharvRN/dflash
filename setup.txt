# DFlash Pod Setup (atharv-rwx-pod)

# 1) Enter pod
kubectl -n wenglab-interpretable-ai exec -it atharv-rwx-pod -- bash

# 2) Create + activate env
source /opt/conda/etc/profile.d/conda.sh
conda create -y -n dflash python=3.12
conda activate dflash

# 3) Core tooling
python -m pip install -U pip setuptools wheel

# 4) Torch stack (working pinned combo)
python -m pip install \
  torch==2.9.1 \
  torchvision==0.24.1 \
  torchaudio==2.9.1

# 5) Prebuilt FlashAttention wheel (avoid source build)
python -m pip install --no-deps \
  "flash-attn @ https://github.com/Dao-AILab/flash-attention/releases/download/v2.8.3/flash_attn-2.8.3%2Bcu12torch2.9cxx11abiTRUE-cp312-cp312-linux_x86_64.whl"

# 6) SGLang + required runtime deps
python -m pip install sglang==0.5.9
python -m pip install imageio diffusers cache_dit av addict

# 7) Common libs used by repo scripts
python -m pip install accelerate rich packaging ninja datasets loguru pandas matplotlib tqdm requests

# 8) Verify
python - <<'PY'
import torch, flash_attn, sglang, flashinfer
print("torch:", torch.__version__)
print("flash_attn:", flash_attn.__version__)
print("sglang:", sglang.__version__)
print("flashinfer:", flashinfer.__version__)
PY

sglang serve --help >/dev/null && echo "sglang CLI OK"
